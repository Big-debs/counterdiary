# 1️⃣ Tailwind Config – Diary Theme Foundation

Extend Tailwind with diary colors, shadows, fonts, and animations.

**tailwind.config.js**

```js
/** @type {import('tailwindcss').Config} */
module.exports = {
  content: ["./*.html", "./src/**/*.{js,ts,jsx,tsx}"],
  theme: {
    extend: {

      colors: {
        paper: "#F4EFE6",
        paperLight: "#FFFDF8",
        ink: "#2B2B2B",
        inkSoft: "#6B5E4A",
        diaryGreen: "#0B6E4F",
        diaryAccent: "#E8DFCF",
      },

      fontFamily: {
        diary: ["Patrick Hand", "cursive"],
        body: ["Inter", "sans-serif"],
      },

      boxShadow: {
        paper: `
          0 1px 0 rgba(0,0,0,0.04),
          0 6px 14px rgba(0,0,0,0.08)
        `,
        paperHover: `
          0 4px 10px rgba(0,0,0,0.12),
          0 10px 22px rgba(0,0,0,0.12)
        `,
      },

      animation: {
        pageIn: "pageIn 0.4s ease",
      },

      keyframes: {
        pageIn: {
          "0%": {
            opacity: "0",
            transform: "translateY(10px) rotate(-0.5deg)",
          },
          "100%": {
            opacity: "1",
            transform: "translateY(0px) rotate(0deg)",
          },
        },
      },

      backgroundImage: {
        "paper-lines": `
          linear-gradient(
            transparent 27px,
            rgba(0,0,0,0.05) 28px
          )
        `,
      },

      backgroundSize: {
        lines: "100% 28px",
      },

    },
  },
  plugins: [],
}
```
